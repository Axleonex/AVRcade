cmake_minimum_required(VERSION 3.24)
project(VRClientFnvNative LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Configure with the Windows x86 toolchain (vcvars32).")
endif()

# ponytail: reuse the vendored decoder, not a new disassembler dependency.
set(root "${CMAKE_CURRENT_LIST_DIR}/../..")
set(hde "${root}/external/cyberpunk-vr-port-reference/externals/minhook/src/hde")
add_executable(fnv-render-observer
  "${root}/tools/native/fnv-render-observer.cpp" "${hde}/hde32.c")
target_include_directories(fnv-render-observer PRIVATE "${hde}")
target_compile_features(fnv-render-observer PRIVATE cxx_std_20)
set_property(TARGET fnv-render-observer PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
  target_compile_options(fnv-render-observer PRIVATE /W4)
endif()
enable_testing()
add_test(NAME fnv-observer-self-test COMMAND fnv-render-observer --self-test)

add_library(fnv-legacy-stereo STATIC "${root}/src/native/runtime/openxr/legacy_stereo.cpp")
target_compile_features(fnv-legacy-stereo PUBLIC cxx_std_20)
target_include_directories(fnv-legacy-stereo PRIVATE "${root}/external/openxr/include")
target_link_libraries(fnv-legacy-stereo PUBLIC d3d9 d3d11 dxgi)
set_property(TARGET fnv-legacy-stereo PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(fnv-runtime-check "${root}/tools/native/fnv-runtime-check.cpp")
target_link_libraries(fnv-runtime-check PRIVATE fnv-legacy-stereo)
set_property(TARGET fnv-runtime-check PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_test(NAME fnv-camera-stereo-check COMMAND fnv-runtime-check --self-test)

add_subdirectory("${root}/external/cyberpunk-vr-port-reference/externals/minhook" minhook)
set_property(TARGET minhook PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(vrclient_fnv_stereo SHARED native_stereo.cpp native_stereo.def)
target_link_libraries(vrclient_fnv_stereo PRIVATE fnv-legacy-stereo minhook)
set_property(TARGET vrclient_fnv_stereo PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(EXISTS "${root}/build/gtasa-x86/openxr_loader.dll")
  configure_file("${root}/build/gtasa-x86/openxr_loader.dll" "${CMAKE_CURRENT_BINARY_DIR}/openxr_loader.dll" COPYONLY)
endif()
add_executable(fnv-test-launcher WIN32 "${root}/tools/native/fnv-test-launcher.cpp")
target_compile_features(fnv-test-launcher PRIVATE cxx_std_20)
target_link_libraries(fnv-test-launcher PRIVATE bcrypt shell32 version)
set_property(TARGET fnv-test-launcher PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
