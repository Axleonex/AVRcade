cmake_minimum_required(VERSION 3.24)
project(AvrcadeControllerRemap LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(avrcade-controller-remap SHARED layer.cpp)
target_include_directories(avrcade-controller-remap PRIVATE "${OPENXR_INCLUDE}")
target_compile_definitions(avrcade-controller-remap PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(avrcade-controller-remap PRIVATE shell32 ole32)
if(CMAKE_SIZEOF_VOID_P EQUAL 4)
  target_link_options(avrcade-controller-remap PRIVATE "/EXPORT:xrNegotiateLoaderApiLayerInterface=_xrNegotiateLoaderApiLayerInterface@12")
endif()
target_compile_options(avrcade-controller-remap PRIVATE /W4 /WX /utf-8)
include(CTest)
if(BUILD_TESTING)
  add_executable(controller-remap-tests tests.cpp)
  target_include_directories(controller-remap-tests PRIVATE "${OPENXR_INCLUDE}")
  target_compile_definitions(controller-remap-tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN AVRCADE_LAYER_TEST)
  target_link_libraries(controller-remap-tests PRIVATE shell32 ole32)
  add_test(NAME per-game-input-rules COMMAND controller-remap-tests)
  add_library(controller-remap-test-layer SHARED layer.cpp)
  target_include_directories(controller-remap-test-layer PRIVATE "${OPENXR_INCLUDE}")
  target_compile_definitions(controller-remap-test-layer PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN AVRCADE_LAYER_TEST)
  target_link_libraries(controller-remap-test-layer PRIVATE shell32 ole32)
  add_library(controller-remap-test-runtime SHARED loader_test_runtime.cpp)
  target_include_directories(controller-remap-test-runtime PRIVATE "${OPENXR_INCLUDE}")
  add_executable(controller-remap-loader-test loader_test.cpp)
  target_include_directories(controller-remap-loader-test PRIVATE "${OPENXR_INCLUDE}")
  if(CMAKE_SIZEOF_VOID_P EQUAL 4)
    target_link_options(controller-remap-test-layer PRIVATE "/EXPORT:xrNegotiateLoaderApiLayerInterface=_xrNegotiateLoaderApiLayerInterface@12")
    target_link_options(controller-remap-test-runtime PRIVATE "/EXPORT:xrNegotiateLoaderRuntimeInterface=_xrNegotiateLoaderRuntimeInterface@8")
  endif()
  file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/fixture/fixture-layer.json" CONTENT "{\"file_format_version\":\"1.0.0\",\"api_layer\":{\"name\":\"XR_APILAYER_AVRCADE_fixture\",\"library_path\":\"$<TARGET_FILE:controller-remap-test-layer>\",\"api_version\":\"1.0\",\"implementation_version\":\"1\",\"description\":\"Test only\"}}")
  file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/fixture-runtime.json" CONTENT "{\"file_format_version\":\"1.0.0\",\"runtime\":{\"library_path\":\"$<TARGET_FILE:controller-remap-test-runtime>\"}}")
  add_test(NAME official-loader-remapping COMMAND controller-remap-loader-test "${OPENXR_LOADER}" "$<TARGET_FILE:controller-remap-test-runtime>")
  set_tests_properties(official-loader-remapping PROPERTIES ENVIRONMENT "XR_RUNTIME_JSON=${CMAKE_BINARY_DIR}/fixture-runtime.json;XR_API_LAYER_PATH=${CMAKE_BINARY_DIR}/fixture;XR_ENABLE_API_LAYERS=XR_APILAYER_AVRCADE_fixture")
endif()
